% ---------- Business understanding: CSF, KPI, SMART ----------
\begin{topic}{1 \; Business understanding: CSF \& KPI}{cKpi}
\insight{\term{CSF} (critical success factor): \emph{what must go right}; key activity, often qualitative $\Rightarrow$ hard to measure itself.}
\insight{\term{KPI}: strategically aligned measure embodying a strategic objective/CSF; measured against a \term{target} with a \term{performance range} (above/on/below), \term{encoding} for visual display (e.g.\ \%), assigned to a \term{time frame}. (A KPI is \emph{not} independent of strategy.)}
\insight{\term{Outcome KPI = lagging}: output of past activity (e.g.\ injuries, pass rate). \term{Driver KPI = leading}: activities that will impact outcome KPIs later.}
\insight{\term{SMART}: Specific, Measurable, Assignable (who), Realistic, Time-related.}
\insight{Full KPI spec: (1) formula, (2) unit, (3) data source, (4) update frequency, (5) measurable target + deadline.}
\formula{Rate KPI}{\frac{\#\text{events}}{\text{exposure}}\times 100}{always use a \term{denominator} (e.g.\ injuries per million cyclist passages): a raw count moves when volume moves.}
\insight{Chart per KPI: trend vs own past $\to$ line chart (not pie: no time); single value vs target $\to$ bar/bullet graph.}
\end{topic}
