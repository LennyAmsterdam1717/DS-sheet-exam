\begin{topic}{1 \; CRISP-DM \& business understanding: CSF, KPI}{cKpi}
\sub{CRISP-DM (6 phases, iterative; quiz activities in bold)}
\insight{\term{Business underst.}: aim, \textbf{determine DM goals}, \textbf{DM success criteria}, plan. \term{Data underst.}: collect, describe, \textbf{EDA}, \textbf{verify data quality}. \term{Data prep.}: select, clean, \textbf{construct data} (derive attrs), integrate, format. \term{Modelling}: technique, test design, build. \term{Evaluation}: \textbf{assess model} (usefulness vs business goals; Modelling also ok). \term{Deployment}: put to use (report, monitor).}
\insight{\term{CSF} (critical success factor): \emph{what must go right}; key activity, often qualitative $\Rightarrow$ hard to measure itself.}
\insight{\term{KPI}: measure of a strategic objective/CSF; measured against a \term{target} with a \term{performance range} (above/on/below), \term{encoding} for visual display (e.g.\ \%), assigned to a \term{time frame}. (A KPI is \emph{not} independent of strategy.)}
\insight{\term{Outcome KPI = lagging}: output of past activity (e.g.\ injuries, pass rate). \term{Driver KPI = leading}: activities that will impact outcome KPIs later.}
\insight{\term{SMART}: Specific, Measurable, Assignable (who), Realistic, Time-related.}
\insight{Chart per KPI: trend vs own past $\to$ line chart (not pie: no time); single value vs target $\to$ bar/bullet graph.}
\end{topic}
