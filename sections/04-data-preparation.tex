% ---------- Data preparation: quality, cleaning, outliers, transformation, integration, reduction ----------
\begin{topic}{3 \; Data preparation \& data quality}{cPrep}
\sub{4 preprocessing steps (Sharda) + example}
\insight{\term{Consolidation}: collect, select relevant, integrate sources (e.g.\ last month's transactions with quantity \& price).}
\insight{\term{Cleaning}: missing values, outliers/noise, duplicates (e.g.\ remove transactions with negative price).}
\insight{\term{Transformation}: data types, normalise ranges (for Euclidean distance), discretise, construct attributes (e.g.\ value = quantity $\cdot$ price; income/age).}
\insight{\term{Reduction}: dimension (PCA), volume (subsample), balancing (few positives).}
\sub{Missing values}
\insight{Options: ignore tuple (ok if class label missing; bad if many), fill manually, global constant (``unknown''), attribute mean, \term{class-conditional mean} (smarter), time series: neighbouring value. Course advice: flag as NaN, impute only when the technique needs it, don't delete instances; flag obvious outliers as NaN too.}
\sub{Outliers}
\insight{\term{Global} (point anomaly), \term{contextual} (20\textdegree C in winter; contextual vs behavioural attributes), \term{collective} (group deviates, e.g.\ DoS packets). Outlier detection $\ne$ novelty detection.}
\insight{Methods: statistical (parametric: low density under fitted $f(x,\theta)$; non-parametric: histogram, KDE), distance-, density-, model-based. Global vs local reference set; labeling (binary) vs scoring (degree).}
\formula{$3\sigma$ rule}{z=\frac{x-\hat\mu}{\hat\sigma},\ \hat\mu=\xbar,\ \hat\sigma^2=\tfrac{1}{n-1}\textstyle\sum(x_i-\xbar)^2}{$|z|>3$ $\Rightarrow$ outlier ($\hat\mu\pm3\hat\sigma$ holds 99.7\% of a normal).}
\insight{The outlier itself inflates $\hat\mu$ and $\hat\sigma$ $\Rightarrow$ its $z$ shrinks; two extremes can hide each other (\term{masking}). Works only for roughly normal data with few outliers; multimodal data $\Rightarrow$ mixture of normals.}
\sub{Normalisation (transformation)}
\formula{Min-max}{v'=\frac{v-\min_A}{\max_A-\min_A}(\text{new}_{\max}-\text{new}_{\min})+\text{new}_{\min}}{linear rescale to a new range, e.g.\ [0,1].}
\formula{Z-score}{v'=\frac{v-\bar A}{\sigma_A}}{mean 0, std 1.}
\formula{Decimal scaling}{v'=\frac{v}{10^j}}{$j$ = smallest integer with $\max|v'|<1$.}
\insight{Also: reformat (Jack Wilsher $\to$ Wilsher, J.), same units (inch/cm, EUR/USD), discretisation, label / one-hot encoding.}
\sub{Integration \& entity resolution}
\insight{Schema integration (cust-id $\equiv$ cust-No), entity identification (Bill = William Clinton), value conflicts (units, scales). Redundant attributes $\to$ correlation/covariance analysis.}
\insight{\term{Entity resolution} (dedup, record linkage): normalise schema \& data, compare features with similarity (Jaro for names, edit distance for typos, Soundex phonetic, Jaccard sets, cosine vectors). Pairwise is $O(n^2)$; \term{blocking} into $k$ blocks $\to O(k(n/k)^2)$.}
\sub{Reduction}
\insight{\term{Curse of dimensionality}: more dims $\to$ sparse data, distances less meaningful; $D$ features split in 2 intervals $\to 2^D$ cells (3 $\to$ 8, 4 $\to$ 16).}
\insight{\term{PCA}: eigenvectors of covariance matrix = new axes capturing most variance; keep first $k$ (eigenvalue $>1$ considered important). Also SVD, feature subset selection: drop redundant (price \& sales tax) / irrelevant (student ID for GPA).}
\insight{Model-based (regression, log-linear), histograms (equal-width vs equal-frequency buckets), \term{sampling}: SRS without/with replacement, \term{stratified} (per partition, proportional; for skewed data), cluster sample.}
\end{topic}
