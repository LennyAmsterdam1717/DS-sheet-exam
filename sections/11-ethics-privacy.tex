\begin{topic}{10 \; Ethics \& privacy (Part F, terms from exercises)}{cEthics}
\insight{Removing names/addresses is \emph{not} anonymisation: combining filters (quasi-identifiers) re-identifies people. Ex: 48 neighbourhoods $\times$ 6 age bands $\times$ 5 household types $\times$ 2 illness = 2\,880 cells; 31\,000/2\,880 $\approx$ 10.8 per cell on average, but the average hides many tiny cells (``Overvecht, 75+, single, ill: $n=2$'').}
\insight{Shadow profiles (Case F): data shared by users reveals people who never consented.}
\insight{Trustworthy AI, 7 requirements: human agency \& oversight; technical robustness \& safety; privacy \& data governance; transparency; diversity, non-discrimination \& fairness; societal \& environmental well-being; accountability.}
\insight{Also know: consequentialism vs categorical moral reasoning (trolley problem); fairness examples dog-vs-wolf, doctor-vs-nurse; k-anonymity, l-diversity, t-closeness, $\epsilon$-differential privacy; GDPR: consent, breach notification, right of access, right to be forgotten, data portability, privacy by design, record keeping \& DPO.}
\end{topic}
