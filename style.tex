% ---------- topic block:  \begin{topic}{Title}{colour} ... \end{topic} ----------
% Coloured header bar + coloured closing rule. Plain flowing text inside, so topics
% break cleanly across columns and pages (framed boxes leave gaps in multicols).
\newenvironment{topic}[2]{%
  \def\tc{#2}\par\needspace{5\baselineskip}\vspace{0.7mm}%
  \noindent\colorbox{#2}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{\color{white}\bfseries\large\strut #1}}%
  \par\nopagebreak\vspace{0.25mm}}%
  {\par\nopagebreak\vspace{0.2mm}{\color{\tc}\hrule height 0.5pt}\vspace{0.4mm}}
\setlength{\fboxsep}{0.6mm}
\def\tc{black}

% ---------- entry macros ----------
% \sub{Heading}                    small coloured heading inside a topic
\newcommand{\sub}[1]{\par\vspace{0.6pt}{\color{\tc}\bfseries #1}\par}
% \formula{Name}{math}{note}       name + inline (display-style) formula; note = what it is / how to read values
\newcommand{\formula}[3]{\par\hangindent=0.9em\hangafter=1%
  \textbf{#1}\ $\displaystyle\sidelimits #2$%
  \if\relax\detokenize{#3}\relax\else\par\hangindent=0.9em\hangafter=0\noindent{\color{\tc}$\triangleright$}\ #3\fi\par}
% \formulaD{Name}{math}{note}      same, but the formula gets its own centred line (for long formulas)
\newcommand{\formulaD}[3]{\par\textbf{#1}\vspace{-1.2pt}\[\sidelimits #2\]\vspace{-1.2pt}%
  \if\relax\detokenize{#3}\relax\else\par\hangindent=0.9em\hangafter=0\noindent{\color{\tc}$\triangleright$}\ #3\fi\par}
% \insight{text}                   one bullet of exam-relevant insight
\newcommand{\insight}[1]{\par\hangindent=0.75em\hangafter=1\makebox[0.75em][l]{\color{\tc}\textbullet}#1\par}
% \ex{text}                        worked example from slides/exercises
\newcommand{\ex}[1]{\par\hangindent=0.75em\hangafter=1\makebox[0.75em][l]{\color{\tc}\textit{e}}\textit{#1}\par}
% \term{word}                      bold key term inside running text
\newcommand{\term}[1]{\textbf{#1}}

% sums/products in formulas get their limits at the side (saves height)
